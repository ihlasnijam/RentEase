═══════════════════════════════════════════════
 FIRESTORE DATABASE STRUCTURE
═══════════════════════════════════════════════

users/
  {uid}/
    name, email, phone, role, createdAt

properties/
  {propertyId}/
    name, address, landlordId, totalUnits
    inviteCode (for tenant to join)

units/
  {unitId}/
    propertyId, unitNumber, tenantId, rentAmount
    leaseStart, leaseEnd, status (occupied/vacant)

maintenanceRequests/
  {requestId}/
    tenantId, tenantName, category, location
    description, priority, status, createdAt
    assignedVendor (optional)

payments/
  {paymentId}/
    tenantId, tenantName, amount, month
    status, paidAt

messages/
  {conversationId}/
    chats/
      {messageId}/
        senderId, senderName, message, timestamp


═══════════════════════════════════════════════
 HOW TO GENERATE APK FOR DEMO
═══════════════════════════════════════════════

1. Build → Generate Signed Bundle/APK
2. Select APK → Next
3. Click "Create new" keystore
4. Fill in keystore details (remember the password!)
5. Key alias: rentease
6. Click OK → Next
7. Select "debug" → Create
8. APK saved in app/release/ folder
9. Copy to phone via USB or WhatsApp it to yourself
10. Install on real phone for demo!

 FIREBASE SECURITY RULES (paste in console)
═══════════════════════════════════════════════

// Firestore Rules → go to Firebase Console → Firestore → Rules
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Users can only read/write their own data
    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }

    // Tenants can read/write their own requests
    // Landlords can read all requests
    match /maintenanceRequests/{requestId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }

    // Payments
    match /payments/{paymentId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }

    // Messages
    match /messages/{conversationId}/chats/{messageId} {
      allow read, write: if request.auth != null;
    }
  }
}
